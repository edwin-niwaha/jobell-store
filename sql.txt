CREATE TABLE finance_branch (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    code VARCHAR(20) NOT NULL UNIQUE,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by_id INTEGER,
    CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL
);

ALTER TABLE finance_branch
    ADD COLUMN IF NOT EXISTS created_by_id INTEGER,
    ADD CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL;
	
	
CREATE TABLE IF NOT EXISTS finance_financialperiod (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'open',
    branch_id BIGINT,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by_id INTEGER,
    CONSTRAINT valid_status CHECK (status IN ('open', 'closed', 'locked')),
    CONSTRAINT fk_branch FOREIGN KEY (branch_id) REFERENCES finance_branch(id) ON DELETE RESTRICT,
    CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_finance_financialperiod_start_date ON finance_financialperiod (start_date DESC);


CREATE TABLE accounts (
    id SERIAL PRIMARY KEY,
    account_name VARCHAR(255) NOT NULL,
    account_type VARCHAR(50) NOT NULL,
    account_number VARCHAR(20) NOT NULL UNIQUE,
    description TEXT,
    balance_type VARCHAR(10) NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'active',
    parent_account_id INTEGER,
    branch_id INTEGER,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    is_contra_asset BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by_id INTEGER,
    CONSTRAINT valid_account_type CHECK (account_type IN ('asset', 'liability', 'equity', 'revenue', 'expense')),
    CONSTRAINT valid_balance_type CHECK (balance_type IN ('debit', 'credit')),
    CONSTRAINT valid_status CHECK (status IN ('active', 'inactive', 'closed')),
    CONSTRAINT fk_parent_account FOREIGN KEY (parent_account_id) REFERENCES accounts(id) ON DELETE SET NULL,
    CONSTRAINT fk_branch FOREIGN KEY (branch_id) REFERENCES finance_branch(id) ON DELETE RESTRICT,
    CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL
);

CREATE INDEX idx_accounts_account_number ON accounts (account_number);


CREATE TABLE finance_journalentry (
    id SERIAL PRIMARY KEY,
    reference_number VARCHAR(100) NOT NULL UNIQUE,
    transaction_date DATE NOT NULL,
    description TEXT,
    source VARCHAR(100),
    payee VARCHAR(100),
    payment_method VARCHAR(20),
    financial_period_id INTEGER NOT NULL,
    branch_id INTEGER,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by_id INTEGER,
    CONSTRAINT valid_payment_method CHECK (payment_method IN ('cash', 'bank_transfer', 'check', 'mobile_payment') OR payment_method IS NULL),
    CONSTRAINT fk_financial_period FOREIGN KEY (financial_period_id) REFERENCES finance_financialperiod(id) ON DELETE RESTRICT,
    CONSTRAINT fk_branch FOREIGN KEY (branch_id) REFERENCES finance_branch(id) ON DELETE RESTRICT,
    CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL
);


CREATE TABLE finance_transaction (
    id SERIAL PRIMARY KEY,
    journal_entry_id INTEGER NOT NULL,
    account_id INTEGER NOT NULL,
    amount NUMERIC(10, 2) NOT NULL,
    transaction_type VARCHAR(10) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by_id INTEGER,
    CONSTRAINT valid_transaction_type CHECK (transaction_type IN ('debit', 'credit')),
    CONSTRAINT fk_journal_entry FOREIGN KEY (journal_entry_id) REFERENCES finance_journalentry(id) ON DELETE CASCADE,
    CONSTRAINT fk_account FOREIGN KEY (account_id) REFERENCES accounts(id) ON DELETE CASCADE,
    CONSTRAINT fk_created_by FOREIGN KEY (created_by_id) REFERENCES auth_user(id) ON DELETE SET NULL
);
